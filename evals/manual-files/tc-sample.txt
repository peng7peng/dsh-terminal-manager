[call函数声明]
call() {
  local s="$1"
  if [[ "$s" == *,* ]]; then
    printf 'call "%s",%s\n' "${s%%,*}" "${s#*,}" | unibsp.out shell
  else
    printf 'call "%s"\n' "$s" | unibsp.out shell
  fi
}
